dashboard you run `sql.run()` in the page and pass the rows the same way.

export const spend = (() => {
  const rows = [];
  const today = new Date();
  for (let i = 13; i >= 0; i -= 1) {
    const date = new Date(today);
    date.setDate(today.getDate() - i);
    const day = date.toISOString().slice(0, 10);
    rows.push({ day, region: 'UK', spend: 1200 + ((i * 137) % 900) });
    rows.push({ day, region: 'US', spend: 1900 + ((i * 211) % 1200) });
  }
  return rows;
})();

## Horizontal bar chart

Set `horizontal` to put the categories on the y-axis. Long labels are shortened
and the full label stays in the tooltip.

<div class="not-content dashtro-example">
  <BarChart
    client:load
    data={usage}
    x="model"
    series={[{ key: 'requests', label: 'Requests' }]}
    title="Requests by model"
    horizontal
    height={300}
  />
</div>

## Filters

Add a `filters` prop to a chart. The controls run in the browser: the query does
not run again. This chart has a select, a date range and a number range.

<div class="not-content dashtro-example">
  <BarChart
    client:load
    data={spend}
    x="day"
    series={[{ key: 'spend', label: 'Spend' }]}
    title="Spend by day"
    valueFormat={{ style: 'currency', currency: 'GBP' }}
    filters={[
      { type: 'select', field: 'region', label: 'Region', defaultValue: 'UK' },
      {
        type: 'dateRange',
        field: 'day',
        label: 'Window',
        presets: [
          { label: '3 days', days: 3 },
          { label: '7 days', days: 7 },
        ],
        defaultValue: 7,
      },
      { type: 'numberRange', field: 'spend', label: 'Spend', defaultMin: 1000 },
    ]}
    height={340}
  />
</div>
